# OpenAPI Class Viewer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a static React website that lets developers browse and drill into OpenAPI model classes as interactive diagrams.

**Architecture:** A Python script parses `api.swagger.yaml` into `model.json` at setup time. A Vite + React 18 app reads that file at build time and renders it as an interactive React Flow diagram. No backend, no runtime data fetching — everything is pre-computed JSON.

**Tech Stack:** Python 3 + PyYAML (parser), React 18, Vite 5, React Flow v11, Vitest + @testing-library/react (frontend tests), pytest (parser tests), plain CSS.

Create `website/scripts/parse-swagger.py`:

```python
import yaml
import json
from pathlib import Path

PRIMITIVE_TYPES = {'string', 'integer', 'number', 'boolean', 'object'}
TYPE_MAP = {
    'string': 'String',
    'integer': 'Integer',
    'number': 'Number',
    'boolean': 'Boolean',
    'object': 'Object',
}

DOMAINS = {
    'Fulfillment':    ['PickJob', 'PickRun', 'PackJob', 'HandoverJob', 'StowJob', 'InboundProcess'],
    'Commerce':       ['Order', 'Article', 'Listing', 'Category', 'Brand'],
    'Infrastructure': ['Facility', 'Carrier', 'LoadUnit', 'InterfacilityConnection'],
    'Returns':        ['ItemReturnJob', 'InboundReceipt'],
    'Configuration':  ['DomainConfiguration', 'GdprConfiguration', 'Fence'],
}


def resolve_ref(ref):
    """'#/components/schemas/Foo' -> 'Foo'"""
    return ref.split('/')[-1]


def classify_field(field_schema, all_schemas):
    """Return (kind, type_name, is_array) for a property schema."""
    is_array = field_schema.get('type') == 'array'
    schema = field_schema.get('items', field_schema) if is_array else field_schema

    # allOf wrapper around a single $ref is common in OAS3
    if 'allOf' in schema and len(schema['allOf']) >= 1:
        schema = schema['allOf'][0]

    if '$ref' in schema:
        name = resolve_ref(schema['$ref'])
        target = all_schemas.get(name, {})
        kind = 'enum' if ('enum' in target) else 'complex'
        return kind, name, is_array

    raw_type = schema.get('type', 'object')
    if raw_type in PRIMITIVE_TYPES:
        return 'primitive', TYPE_MAP.get(raw_type, 'Object'), is_array

    return 'primitive', 'Object', is_array


def collect_properties(schema, all_schemas, _seen=None):
    """Flatten properties from direct properties + allOf refs."""
    if _seen is None:
        _seen = set()
    props = {}
    name = schema.get('title', '')
    if name in _seen:
        return props
    _seen.add(name)

    if 'properties' in schema:
        props.update(schema['properties'])

    for part in schema.get('allOf', []):
        if '$ref' in part:
            ref_name = resolve_ref(part['$ref'])
            ref_schema = all_schemas.get(ref_name, {})
            props.update(collect_properties(ref_schema, all_schemas, _seen))
        elif 'properties' in part:
            props.update(part['properties'])

    return props


def parse_schemas(schemas):
    classes = {}

    for name, schema in schemas.items():
        # Top-level enum
        if 'enum' in schema:
            classes[name] = {
                'kind': 'enum',
                'values': [v for v in schema['enum'] if v is not None],
                'description': schema.get('description', ''),
            }
            continue

        props = collect_properties(schema, schemas)
        if not props and schema.get('type') not in ('object', None) and 'allOf' not in schema:
            continue  # not a class-like schema

        fields = []
        for field_name, field_schema in props.items():
            kind, type_name, is_array = classify_field(field_schema, schemas)
            fields.append({
                'name': field_name,
                'type': type_name,
                'kind': kind,
                'array': is_array,
            })

        classes[name] = {
            'description': schema.get('description', schema.get('title', '')),
            'fields': fields,
        }

    return classes


def assign_domains(class_names):
    assigned = set()
    domains = {}
    for domain, names in DOMAINS.items():
        domains[domain] = [n for n in names if n in class_names]
        assigned.update(domains[domain])
    domains['Common'] = sorted(c for c in class_names if c not in assigned)
    return domains


def main():
    script_dir = Path(__file__).parent
    swagger_path = script_dir.parent.parent / 'api.swagger.yaml'
    output_path = script_dir.parent / 'src' / 'data' / 'model.json'

    with open(swagger_path) as f:
        swagger = yaml.safe_load(f)

    schemas = swagger.get('components', {}).get('schemas', {})
    classes = parse_schemas(schemas)
    domains = assign_domains(set(classes.keys()))

    model = {'classes': classes, 'domains': domains}

    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, 'w') as f:
        json.dump(model, f, indent=2)

    print(f"Wrote {len(classes)} classes to {output_path}")


if __name__ == '__main__':
    main()
```

import { useState, useCallback } from 'react'
import { useNodesState, useEdgesState } from 'reactflow'

export function useDiagram(classes) {
  const [nodes, setNodes, onNodesChange] = useNodesState([])
  const [edges, setEdges, onEdgesChange] = useEdgesState([])
  const [breadcrumb, setBreadcrumb] = useState([])

  const openEntity = useCallback((name) => {
    const classData = classes[name]
    if (!classData) return
    setNodes([{
      id: name,
      type: classData.kind === 'enum' ? 'enumNode' : 'classNode',
      data: { name, ...classData, role: 'focus', collapsed: false },
      position: { x: 0, y: 0 },
    }])
    setEdges([])
    setBreadcrumb([name])
  }, [classes, setNodes, setEdges])

  const drillInto = useCallback((sourceName, fieldName, targetName) => {
    const classData = classes[targetName]
    if (!classData) return
    const isEnum = classData.kind === 'enum'
    const edgeId = `${sourceName}-${fieldName}-${targetName}`

    setEdges(prev => {
      if (prev.find(e => e.id === edgeId)) return prev
      return [...prev, {
        id: edgeId,
        source: sourceName,
        target: targetName,
        type: 'smoothstep',
        style: {
          stroke: isEnum ? '#d97706' : '#6366f1',
          ...(isEnum ? { strokeDasharray: '5,5' } : {}),
        },
      }]
    })

    setNodes(prev => {
      if (prev.find(n => n.id === targetName)) return prev
      return [...prev, {
        id: targetName,
        type: isEnum ? 'enumNode' : 'classNode',
        data: { name: targetName, ...classData, role: 'related', collapsed: false },
        position: { x: prev.length * 340, y: 160 },
      }]
    })

    setBreadcrumb(prev => prev.includes(targetName) ? prev : [...prev, targetName])
  }, [classes, setNodes, setEdges])

  const navigateTo = useCallback((name) => {
    openEntity(name)
  }, [openEntity])

  const toggleCollapse = useCallback((nodeId) => {
    setNodes(prev => prev.map(n =>
      n.id === nodeId ? { ...n, data: { ...n.data, collapsed: !n.data.collapsed } } : n
    ))
  }, [setNodes])

  const resetLayout = useCallback(() => {
    setNodes(prev => prev.map((n, i) => ({ ...n, position: { x: i * 340, y: 0 } })))
  }, [setNodes])

  return {
    nodes, edges, breadcrumb,
    onNodesChange, onEdgesChange,
    openEntity, drillInto, navigateTo, toggleCollapse, resetLayout,
    setNodes, setEdges,
  }
}

import { useState } from 'react'

export function Sidebar({ domains, classes, activeEntity, onSelect }) {
  const [search, setSearch] = useState('')
  const [collapsed, setCollapsed] = useState({})

  const toggleDomain = (domain) =>
    setCollapsed(prev => ({ ...prev, [domain]: !prev[domain] }))

  const matches = (name) => !search || name.toLowerCase().includes(search.toLowerCase())

  return (
    <div style={{
      width: 220,
      flexShrink: 0,
      background: '#12121f',
      display: 'flex',
      flexDirection: 'column',
      height: '100vh',
      overflow: 'hidden',
    }}>
      <div style={{ padding: '10px 12px' }}>
        <input
          value={search}
          onChange={e => setSearch(e.target.value)}
          placeholder="Search entities…"
          style={{
            width: '100%',
            background: '#1e1e2e',
            border: '1px solid #2d2d44',
            borderRadius: 4,
            padding: '5px 8px',
            color: '#cbd5e1',
            fontSize: 12,
            boxSizing: 'border-box',
          }}
        />
      </div>
      <div style={{ overflowY: 'auto', flex: 1 }}>
        {Object.entries(domains).map(([domain, names]) => {
          const filtered = names.filter(matches)
          if (search && filtered.length === 0) return null
          return (
            <div key={domain}>
              <div
                onClick={() => toggleDomain(domain)}
                style={{
                  padding: '6px 12px',
                  color: '#64748b',
                  fontSize: 11,
                  fontWeight: 'bold',
                  textTransform: 'uppercase',
                  cursor: 'pointer',
                  letterSpacing: '0.05em',
                  userSelect: 'none',
                }}
              >
                {collapsed[domain] ? '▶' : '▼'} {domain}
              </div>
              {!collapsed[domain] && filtered.map(name => {
                const classData = classes[name]
                const nonPrimCount = classData?.fields?.filter(f => f.kind === 'complex').length ?? 0
                const isActive = name === activeEntity
                return (
                  <div
                    key={name}
                    onClick={() => onSelect(name)}
                    style={{
                      padding: '4px 12px 4px 20px',
                      cursor: 'pointer',
                      fontSize: 12,
                      color: isActive ? '#a78bfa' : '#cbd5e1',
                      borderLeft: isActive ? '2px solid #6366f1' : '2px solid transparent',
                      background: isActive ? '#1e1e2e' : 'transparent',
                      display: 'flex',
                      justifyContent: 'space-between',
                    }}
                  >
                    <span>{name}</span>
                    {nonPrimCount > 0 && (
                      <span style={{ color: '#64748b', fontSize: 10 }}>{nonPrimCount}</span>
                    )}
                  </div>
                )
              })}
            </div>
          )
        })}
      </div>
    </div>
  )
}

      />
    </div>
  )
}
